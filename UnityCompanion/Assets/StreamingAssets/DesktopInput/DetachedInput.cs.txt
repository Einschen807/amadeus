 void Apply(Message m){
  switch(m.op){
   case "text":updating=true;editor.Text=m.value??"";editor.SelectionStart=editor.TextLength;updating=false;break;
   case "visible":wanted=m.enabled;Follow();break;
   case "enabled":allowSend=m.enabled;send.Enabled=allowSend;break;
   case "attachment":attachment=m.value??"";attachmentLabel.Text=attachment;LayoutInput();Follow();break;
   case "focus":if(wanted){Follow();SetForegroundWindow(Handle);editor.Focus();}break;
   case "quit":closing=true;Close();break;
   case "probe":Rect r;GetWindowRect(owner.Handle,out r);Write(new{op="probe",visible=Visible,value=editor.Text,attachment=attachment,x=Left,y=Top,width=Width,height=Height,ownerBottom=r.bottom,ownerLeft=r.left,ownerWidth=r.right-r.left,owner=owner.Handle.ToInt64(),handle=Handle.ToInt64()});break;

 void Follow(){
  if(!IsWindow(owner.Handle)){closing=true;Close();return;}
  if(!wanted||IsIconic(owner.Handle)||!IsWindowVisible(owner.Handle)){if(Visible)Hide();return;}
  Rect r;if(!GetWindowRect(owner.Handle,out r))return;
  var area=Screen.FromHandle(owner.Handle).WorkingArea;int h=string.IsNullOrEmpty(attachment)?90:118;
  int mainHeight=r.bottom-r.top;
  // Move the fixed-size phone upward if that leaves room for the input below it.
  if(r.bottom+6+h>area.Bottom){
   int y=Math.Max(area.Top,area.Bottom-mainHeight-h-6);
   if(y<r.top)SetWindowPos(owner.Handle,IntPtr.Zero,r.left,y,0,0,0x15);
   GetWindowRect(owner.Handle,out r);
  }
  int width=Math.Min(r.right-r.left,area.Width);int left=Math.Max(area.Left,Math.Min(r.left,area.Right-width));
  int top=r.bottom+6;
  // When the screen cannot fit both vertically, dock the input beside the phone.
  if(top+h>area.Bottom){
   if(r.right+6+width<=area.Right)left=r.right+6;
   else if(r.left-width-6>=area.Left)left=r.left-width-6;
   else left=Math.Max(area.Left,Math.Min(r.left,area.Right-width));
   top=Math.Max(area.Top,Math.Min(r.top,area.Bottom-h));
  }
  SetBounds(left,top,width,h);if(!Visible)Show(owner);
 }
